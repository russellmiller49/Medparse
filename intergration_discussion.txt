# src/graph/umls_subset_loader.py
import pandas as pd
from neo4j import GraphDatabase

# RRF column names (UMLS RRF spec)
MRCONSO_COLS = ["CUI","LAT","TS","LUI","STT","SUI","ISPREF","AUI","SAUI","SCUI","SDUI","SAB","TTY","CODE","STR","SRL","SUPPRESS","CVF"]
MRREL_COLS   = ["CUI1","AUI1","STYPE1","REL","CUI2","AUI2","STYPE2","RELA","RUI","SRUI","SAB","SL","RG","DIR","SUPPRESS","CVF"]
MRSTY_COLS   = ["CUI","TUI","STN","STY","ATUI","CVF"]

def load_umls_to_neo4j(path_rrf_dir, neo4j_uri, auth):
    conso = pd.read_csv(f"{path_rrf_dir}/MRCONSO.RRF", sep="|", names=MRCONSO_COLS, dtype=str, engine="python")
    rel   = pd.read_csv(f"{path_rrf_dir}/MRREL.RRF",   sep="|", names=MRREL_COLS,   dtype=str, engine="python")
    sty   = pd.read_csv(f"{path_rrf_dir}/MRSTY.RRF",   sep="|", names=MRSTY_COLS,   dtype=str, engine="python")

    # filter SAB and suppressibles
    conso = conso[(conso["SAB"].isin(["SNOMEDCT_US","ICD10CM","CPT","LOINC","RXNORM","MSH"])) & (conso["SUPPRESS"] != "O")]
    # pivot synonyms
    pref = conso.sort_values("ISPREF", ascending=False).groupby("CUI").first()["STR"].rename("pref_name")
    aliases = conso.groupby("CUI")["STR"].apply(lambda s: list({x for x in s if x})).rename("aliases")
    codes = conso.groupby("CUI").apply(lambda df: df[["SAB","CODE"]].dropna().drop_duplicates().to_dict("records")).rename("sab_codes")

    # attach semantic type
    tui = sty.groupby("CUI")["STY"].apply(lambda s: list(sorted(set(s)))).rename("semantic_types")

    df = pd.concat([pref, aliases, codes, tui], axis=1).reset_index()

    driver = GraphDatabase.driver(neo4j_uri, auth=auth)
    with driver.session() as sess:
        sess.run("CREATE CONSTRAINT concept_cui IF NOT EXISTS FOR (c:Concept) REQUIRE c.cui IS UNIQUE")
        query = """
        UNWIND $rows AS row
        MERGE (c:Concept {cui: row.CUI})
        SET c.pref_name = row.pref_name,
            c.aliases   = row.aliases,
            c.semantic_type = row.semantic_types
        WITH c,row
        UNWIND row.sab_codes AS sc
        MERGE (c)-[:HAS_CODE {sab: sc.SAB}]->(:Code {code: sc.CODE, sab: sc.SAB})
        """
        sess.run(query, parameters={"rows": df.to_dict("records")})

        # relationships
        rel = rel[rel["REL"].isin(["PAR","CHD","RB","RN","RO"]) & (rel["SUPPRESS"]!="Y")]
        mapping = {"PAR":"IS_A", "CHD":"IS_A", "RB":"IS_A", "RN":"IS_A", "RO":"RELATED_TO"}
        rel["TYPE"] = rel["REL"].map(mapping)
        sess.run("""
        UNWIND $rows AS r
        MATCH (a:Concept {cui: r.CUI1}), (b:Concept {cui: r.CUI2})
        MERGE (a)-[e:REL {rel:r.REL, type:r.TYPE, sab:r.SAB}]->(b)
        """, parameters={"rows": rel[["CUI1","CUI2","REL","SAB"]].to_dict("records")})

8) Query patterns you’ll use (Cypher)

Map free‑text query → CUIs → expand 1–2 hops (filtered by semantic type):

// Assume you already resolved the text to top CUIs: $cuis
MATCH (c:Concept) WHERE c.cui IN $cuis
CALL apoc.path.expandConfig(c, {
  relationshipFilter: "REL|IS_A|RELATED_TO>",
  minLevel: 1, maxLevel: 2
}) YIELD path
WITH DISTINCT nodes(path) AS ns
UNWIND ns AS n
WITH DISTINCT n WHERE n:Concept AND any(t IN n.semantic_type WHERE t IN $keep_tuis)
RETURN n.cui AS cui, n.pref_name AS term, n.semantic_type AS tuis
LIMIT 200


Collect supporting documents for a bag of CUIs:

UNWIND $cuis AS cui
MATCH (d:Document)-[m:MENTIONS]->(c:Concept {cui:cui})
RETURN d.pmid AS pmid, d.title AS title, count(m) AS hits
ORDER BY hits DESC
LIMIT 50

# shared_pipeline.py
class MedicalDataPipeline:
    def __init__(self):
        self.medparse = MedparseConnector()
        self.qdrant = QdrantClient()
        self.kg_builder = KnowledgeGraphBuilder()
        
    def process_new_document(self, pdf_path, doc_type='guideline'):
        """
        1. Extract with Medparse
        2. Build knowledge graph entries
        3. Store in Qdrant
        4. Generate questions if appropriate
        """
        
        # Step 1: Extract structured data
        extracted = self.medparse.ingest_guideline(pdf_path)
        
        # Step 2: Build knowledge graph from UMLS entities
        for entity in extracted['umls_entities']:
            self.kg_builder.add_umls_concept(
                cui=entity['cui'],
                text=entity['text'],
                semantic_types=entity['tui']
            )
        
        # Step 3: Vectorize and store
        doc_chunks = self.chunk_for_rag(extracted)
        embeddings = self.embed_chunks(doc_chunks)
        self.qdrant.upsert(collection="medical_kb", points=embeddings)
        
        # Step 4: Generate questions from key findings
        if doc_type == 'guideline':
            questions = self.generate_questions_from_stats(extracted['statistics'])
            return questions

# ip_assist_lite/src/generation/enhanced_generator.py
class EnhancedQuestionGenerator:
    def __init__(self, medparse_connector):
        self.medparse = medparse_connector
        
    def generate_from_research_paper(self, pdf_path):
        """Generate questions from research papers"""
        
        # Extract using Medparse
        paper_data = self.medparse.ingest_guideline(pdf_path)
        
        # Create questions from different components
        questions = []
        
        # 1. Statistics-based questions
        for stat in paper_data['statistics']:
            if stat['type'] == 'hazard_ratio':
                q = self.create_hr_interpretation_question(stat)
                questions.append(q)
        
        # 2. Figure interpretation questions
        for figure in paper_data['figures']:
            if 'kaplan_meier' in figure['caption'].lower():
                q = self.create_survival_curve_question(figure)
                questions.append(q)
        
        # 3. UMLS concept-based questions
        disease_concepts = [e for e in paper_data['umls_entities'] 
                          if 'T047' in e['tui']]  # Disease concepts
        
        for disease in disease_concepts:
            q = self.create_disease_management_question(disease)
            questions.append(q)
        
        return questions

# IP_assist_lite/src/graph/medparse_ingest.py
from neo4j import GraphDatabase

def ingest_extraction(neo4j_uri, auth, extraction: dict):
    driver = GraphDatabase.driver(neo4j_uri, auth=auth)
    with driver.session() as s:
        # Document
        s.run("""
        MERGE (d:Document {doc_id:$doc_id})
        SET d += $meta
        """, doc_id=extraction["doc_id"], meta=extraction.get("metadata", {}))

        # UMLs mentions -> Concept + MENTIONS edges with offsets
        s.run("""
        UNWIND $links AS e
        MERGE (c:Concept {cui:e.cui})
        ON CREATE SET c.aliases = [e.text]
        MERGE (d:Document {doc_id:$doc_id})-[:MENTIONS {tui:e.tui, score:e.score, offsets:e.offsets}]->(c)
        """, doc_id=extraction["doc_id"], links=extraction["umls_links"])

        # Figures / Tables
        s.run("""
        UNWIND $figs AS v
        MERGE (va:VisualAsset {id: $doc_id + "::" + coalesce(v.local_id, v.caption)})
        SET va.kind = v.kind, va.caption = v.caption, va.ocr_text = v.ocr_text, va.path = v.image_path
        MERGE (d:Document {doc_id:$doc_id})-[:HAS_FIGURE]->(va)
        """, doc_id=extraction["doc_id"], figs=[{**f, "kind":"figure"} for f in extraction["figures"]])

        s.run("""
        UNWIND $tabs AS v
        MERGE (va:VisualAsset {id: $doc_id + "::" + coalesce(v.local_id, v.caption)})
        SET va.kind = v.kind, va.caption = v.caption, va.ocr_text = v.ocr_text
        MERGE (d:Document {doc_id:$doc_id})-[:HAS_TABLE]->(va)
        """, doc_id=extraction["doc_id"], tabs=[{**t, "kind":"table"} for t in extraction["tables"]])

        # Stats
        s.run("""
        UNWIND $stats AS st
        MERGE (o:StatObs {id: $doc_id + "::" + st.kind + "::" + toString(st.value) + "::" + toString(coalesce(st.p_value,''))})
        SET o += st
        MERGE (d:Document {doc_id:$doc_id})-[:EVIDENCE_IN]->(o)
        """, doc_id=extraction["doc_id"], stats=extraction["statistics"])

from neo4j import GraphDatabase

def ensure_constraints(uri, auth):
    q = [
        "CREATE CONSTRAINT concept_cui IF NOT EXISTS FOR (c:Concept) REQUIRE c.cui IS UNIQUE",
        "CREATE CONSTRAINT doc_id IF NOT EXISTS FOR (d:Document) REQUIRE d.doc_id IS UNIQUE",
        "CREATE CONSTRAINT visual_id IF NOT EXISTS FOR (v:VisualAsset) REQUIRE v.id IS UNIQUE",
        "CREATE CONSTRAINT stat_id IF NOT EXISTS FOR (s:StatObs) REQUIRE s.id IS UNIQUE",
        "CREATE INDEX concept_semantic_type IF NOT EXISTS FOR (c:Concept) ON (c.semantic_type)"
    ]
    driver = GraphDatabase.driver(uri, auth=auth)
    with driver.session() as s:
        for stmt in q: s.run(stmt)
    driver.close()


src/graph/umls_subset_loader.py (skeleton)

import pandas as pd
from neo4j import GraphDatabase

MRCONSO = ["CUI","LAT","TS","LUI","STT","SUI","ISPREF","AUI","SAUI","SCUI","SDUI","SAB","TTY","CODE","STR","SRL","SUPPRESS","CVF"]
MRREL   = ["CUI1","AUI1","STYPE1","REL","CUI2","AUI2","STYPE2","RELA","RUI","SRUI","SAB","SL","RG","DIR","SUPPRESS","CVF"]
MRSTY   = ["CUI","TUI","STN","STY","ATUI","CVF"]

def load_umls_subset(rrf_dir: str, keep_sab: set, uri: str, auth):
    conso = pd.read_csv(f"{rrf_dir}/MRCONSO.RRF", sep="|", names=MRCONSO, dtype=str, engine="python")
    rel   = pd.read_csv(f"{rrf_dir}/MRREL.RRF",   sep="|", names=MRREL,   dtype=str, engine="python")
    sty   = pd.read_csv(f"{rrf_dir}/MRSTY.RRF",   sep="|", names=MRSTY,   dtype=str, engine="python")

    conso = conso[(conso["SAB"].isin(keep_sab)) & (conso["SUPPRESS"] != "O")]
    pref = conso.sort_values("ISPREF", ascending=False).groupby("CUI").first()["STR"].rename("pref_name")
    aliases = conso.groupby("CUI")["STR"].apply(lambda s: list({x for x in s if isinstance(x, str)}))
    tui = sty.groupby("CUI")["STY"].apply(lambda s: list(sorted(set(s)))).rename("semantic_type")

    df = pd.concat([pref, aliases, tui], axis=1).reset_index().rename(columns={"index":"CUI"}).fillna({"aliases": []})

    driver = GraphDatabase.driver(uri, auth=auth)
    with driver.session() as s:
        s.run("CREATE CONSTRAINT concept_cui IF NOT EXISTS FOR (c:Concept) REQUIRE c.cui IS UNIQUE")
        s.run("""
        UNWIND $rows AS row
        MERGE (c:Concept {cui: row.CUI})
        SET c.pref_name = row.pref_name,
            c.aliases = row.aliases,
            c.semantic_type = row.semantic_type
        """, rows=df.to_dict("records"))

        rel = rel[rel["REL"].isin(["PAR","CHD","RB","RN","RO"]) & (rel["SUPPRESS"]!="Y")]
        mapping = {"PAR":"IS_A","CHD":"IS_A","RB":"IS_A","RN":"IS_A","RO":"RELATED_TO"}
        records = [{"CUI1": r.CUI1, "CUI2": r.CUI2, "TYPE": mapping.get(r.REL,"RELATED_TO"), "SAB": r.SAB}
                   for r in rel.itertuples()]
        s.run("""
        UNWIND $rows AS r
        MATCH (a:Concept {cui:r.CUI1}), (b:Concept {cui:r.CUI2})
        MERGE (a)-[:REL {type:r.TYPE, sab:r.SAB}]->(b)
        """, rows=records)
    driver.close()

    with driver.session() as s:
        s.run("MERGE (d:Document {doc_id:$id}) SET d += $meta", id=doc_id, meta=ex.get("metadata", {}))

        s.run("""
        UNWIND $links AS e
        MERGE (c:Concept {cui:e.cui})
        MERGE (d:Document {doc_id:$doc})-[:MENTIONS {tui:e.tui, score:e.score, offsets:e.offsets}]->(c)
        """, links=ex["umls_links"], doc=doc_id)

        figs = [{**f, "kind":"figure"} for f in ex.get("figures", [])]
        tabs = [{**t, "kind":"table"} for t in ex.get("tables", [])]

        s.run("""
        UNWIND $assets AS v
        MERGE (va:VisualAsset {id: $doc + "::" + coalesce(v.local_id, substring(v.caption,0,64))})
        SET va.kind = v.kind, va.caption = v.caption, va.ocr_text = v.ocr_text, va.path = v.image_path
        MERGE (d:Document {doc_id:$doc})-[:HAS_FIGURE]->(va)
        """, assets=figs, doc=doc_id)

        s.run("""
        UNWIND $assets AS v
        MERGE (va:VisualAsset {id: $doc + "::" + coalesce(v.local_id, substring(v.caption,0,64))})
        SET va.kind = v.kind, va.caption = v.caption, va.ocr_text = v.ocr_text
        MERGE (d:Document {doc_id:$doc})-[:HAS_TABLE]->(va)
        """, assets=tabs, doc=doc_id)

        s.run("""
        UNWIND $stats AS st
        MERGE (o:StatObs {id: $doc + "::" + st.kind + "::" + toString(st.value) + "::" + toString(coalesce(st.p_value,''))})
        SET o += st
        MERGE (d:Document {doc_id:$doc})-[:EVIDENCE_IN]->(o)
        """, stats=ex.get("statistics", []), doc=doc_id)

7) Graph‑augmented retrieval glue

src/graph/graphrag.py

from neo4j import GraphDatabase
from typing import List, Dict

class GraphRAG:
    def __init__(self, uri, auth, hops=2, prefer_tuis=None):
        self.driver = GraphDatabase.driver(uri, auth=auth)
        self.hops = hops
        self.prefer_tuis = set(prefer_tuis or [])

    def expand_cuis(self, seed_cuis: List[str], limit=200) -> List[Dict]:
        q = """
        MATCH (c:Concept) WHERE c.cui IN $cuis
        CALL apoc.path.expandConfig(c, {relationshipFilter:'REL>', minLevel:1, maxLevel:$hops})
        YIELD path
        WITH DISTINCT last(nodes(path)) AS n
        WHERE n:Concept
        RETURN n.cui AS cui, n.pref_name AS term, n.semantic_type AS tuis
        LIMIT $limit
        """
        with self.driver.session() as s:
            return s.run(q, cuis=seed_cuis, hops=self.hops, limit=limit).data()

    def docs_for_cuis(self, cuis: List[str], k=100) -> List[Dict]:
        q = """
        UNWIND $cuis AS cui
        MATCH (d:Document)-[m:MENTIONS]->(c:Concept {cui:cui})
        RETURN d.doc_id AS id, count(m) AS hits
        ORDER BY hits DESC
        LIMIT $k
        """
        with self.driver.session() as s:
            return s.run(q, cuis=cuis, k=k).data()

    def evidence_signals(self, cuis: List[str]) -> Dict[str, int]:
        q = """
        UNWIND $cuis AS cui
        MATCH (d:Document)-[:EVIDENCE_IN]->(s:StatObs)
        WITH d, count(s) AS stat_count
        MATCH (d)-[:HAS_FIGURE|:HAS_TABLE]->(:VisualAsset)
        WITH d, stat_count, count(*) AS asset_count
        RETURN d.doc_id AS id, stat_count, asset_count
        """
        with self.driver.session() as s:
            return {r["id"]: (r["stat_count"]*2 + r["asset_count"]) for r in s.run(q, cuis=cuis)}

from pydantic import BaseModel, Field
from typing import List, Optional, Dict, Any

class UmlsLink(BaseModel):
    cui: str

    score: float

    p_value: Optional[float] = None
    group_a: Optional[str] = None

class VisualAsset(BaseModel):

from fastapi import APIRouter, UploadFile, Form, HTTPException
from fastapi import Depends
from ..models import ExtractionResult
from ..service import run_full_extraction
from ..config import settings
from ..deps import api_key_required

router = APIRouter(tags=["extract"])

@router.post("/extract", response_model=ExtractionResult, dependencies=[Depends(api_key_required)])
async def extract(pdf: UploadFile, doc_id: str = Form(...)):
    if pdf.content_type not in {"application/pdf"}:
        raise HTTPException(415, "Only application/pdf supported")
    content = await pdf.read()
    max_bytes = settings.MAX_UPLOAD_MB * 1024 * 1024
    if len(content) > max_bytes:
        raise HTTPException(413, f"File too large (> {settings.MAX_UPLOAD_MB} MB)")
    return run_full_extraction(doc_id=doc_id, pdf_bytes=content)

4) CI/CD (GitHub Actions)

Add one workflow in each repo: .github/workflows/ci.yml

name: CI
on:
  push: { branches: [ "**" ] }
  pull_request:

jobs:
  build-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.11" }
      - name: Install
        run: |
          python -m pip install --upgrade pip
          pip install -e ".[dev]" || pip install -r requirements.txt
      - name: Lint & Typecheck
        run: |
          ruff check .
          ruff format --check .
          mypy .
      - name: Tests
        run: pytest -q

  docker:
    needs: build-test
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: docker/setup-buildx-action@v3
      - uses: docker/login-action@v3
        with:
          username: ${{ secrets.DOCKERHUB_USER }}
          password: ${{ secrets.DOCKERHUB_TOKEN }}
      - name: Build & push image
        run: |
          docker build -t ${{ secrets.DOCKERHUB_USER }}/ip-assist-lite:${{ github.sha }} .
          docker push ${{ secrets.DOCKERHUB_USER }}/ip-assist-lite:${{ github.sha }}


(Replicate in Medparse with image name medparse-api.)
